Arrow C Data Interface ABI Structure

The C Data Interface enables 0ms in-process zero-copy data exchange across C++, Rust, Python, DuckDB, Polars, and PyTorch using two ABI-stable structs:

struct ArrowSchema { const char* format; // Type string ("i", "l", "f", "g", "u", "+s") const char* name; // Column name const char* metadata; // Key-value metadata int64_t flags; // Nullable, ordered, etc. int64_t n_children; struct ArrowSchema** children; void (*release)(struct ArrowSchema*); }; struct ArrowArray { int64_t length; // Number of elements int64_t null_count; // Count of nulls int64_t offset; // Zero-copy slice offset int64_t n_buffers; // Buffer count (2 for primitives, 3 for strings) const void** buffers; // Array of pointers to physical memory buffers void (*release)(struct ArrowArray*); };
